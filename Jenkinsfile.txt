pipeline {
    agent any

    environment {
        DOTNET_CLI_HOME = "C:\\Program Files\\dotnet"
        APP_POOL_NAME = "RI_TRACKING_BACKEND"
        WEBSITE_NAME = "RI_TRACKING_BACKEND_DEV"
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build') {
            steps {
                script {
                    bat "dotnet restore"
                    bat "dotnet build --configuration Release --no-restore"
                }
            }
        }

        stage('Test') {
            steps {
                script {
                    bat "dotnet test --no-restore --configuration Release"
                }
            }
        }

        stage('Publish') {
            steps {
                script {
                    bat "if exist .\\publish rmdir /s /q .\\publish"
                    bat "dotnet publish --no-restore --configuration Release --output .\\publish"
                }
            }
        }
        
        stage('Deploy') {
            steps {
                script {
                    powershell '''
                        Write-Host "========================================"
                        Write-Host "Starting Deployment Process"
                        Write-Host "========================================"
                        
                        $appPoolName = $env:APP_POOL_NAME
                        $websiteName = $env:WEBSITE_NAME
                        $destination = "D:\\IIS_Root\\RI_TRACKING_DEV\\RI_TRACKING_BACKEND"
                        $publishFolder = ".\\publish"
                        $maxRetries = 10
                        $lockReleaseWait = 30
                        
                        # Function to get IIS site ID
                        function Get-IISSiteID {
                            param($websiteName)
                            try {
                                Import-Module WebAdministration -ErrorAction Stop
                                $site = Get-Website -Name $websiteName -ErrorAction SilentlyContinue
                                if ($site) {
                                    return $site.ID
                                }
                                return $null
                            } catch {
                                return $null
                            }
                        }
                        
                        # Function to force stop application pool
                        function Stop-IISAppPoolForce {
                            param($appPoolName)
                            
                            try {
                                Import-Module WebAdministration -ErrorAction Stop
                                
                                if (Test-Path "IIS:\\AppPools\\$appPoolName") {
                                    Write-Host "Stopping application pool: $appPoolName"
                                    
                                    # Stop the pool
                                    Stop-WebAppPool -Name $appPoolName -ErrorAction Stop
                                    
                                    # Wait and verify
                                    $timeout = 120
                                    $counter = 0
                                    $poolState = Get-WebAppPoolState -Name $appPoolName
                                    
                                    while ($poolState.Value -ne "Stopped" -and $counter -lt $timeout) {
                                        Start-Sleep -Seconds 2
                                        $counter += 2
                                        $poolState = Get-WebAppPoolState -Name $appPoolName
                                        Write-Host "Waiting for pool to stop... Current state: $($poolState.Value)"
                                    }
                                    
                                    # If still not stopped, try recycle
                                    if ($poolState.Value -ne "Stopped") {
                                        Write-Host "Pool not stopping, attempting recycle..."
                                        Restart-WebAppPool -Name $appPoolName -ErrorAction SilentlyContinue
                                        Start-Sleep -Seconds 10
                                        $poolState = Get-WebAppPoolState -Name $appPoolName
                                    }
                                    
                                    if ($poolState.Value -eq "Stopped") {
                                        Write-Host "Application pool stopped successfully!"
                                        return $true
                                    } else {
                                        Write-Host "Warning: Application pool state is $($poolState.Value)" -ForegroundColor Yellow
                                        return $false
                                    }
                                } else {
                                    Write-Host "Warning: Application pool '$appPoolName' not found" -ForegroundColor Yellow
                                    return $true
                                }
                            } catch {
                                Write-Host "Warning: Could not stop application pool: $_" -ForegroundColor Yellow
                                
                                # Try using appcmd as fallback
                                try {
                                    Write-Host "Attempting to stop pool using appcmd..."
                                    $result = & appcmd stop apppool /apppool.name:"$appPoolName" 2>&1
                                    Write-Host "appcmd result: $result"
                                    Start-Sleep -Seconds 10
                                    return $true
                                } catch {
                                    Write-Host "appcmd also failed: $_" -ForegroundColor Yellow
                                    return $false
                                }
                            }
                        }
                        
                        # Function to start IIS Application Pool
                        function Start-IISAppPool {
                            param($appPoolName)
                            
                            try {
                                if (Test-Path "IIS:\\AppPools\\$appPoolName") {
                                    Write-Host "Starting application pool: $appPoolName"
                                    Start-WebAppPool -Name $appPoolName -ErrorAction Stop
                                    
                                    # Verify pool started
                                    $timeout = 60
                                    $counter = 0
                                    $poolState = Get-WebAppPoolState -Name $appPoolName
                                    while ($poolState.Value -ne "Started" -and $counter -lt $timeout) {
                                        Start-Sleep -Seconds 2
                                        $counter += 2
                                        $poolState = Get-WebAppPoolState -Name $appPoolName
                                        Write-Host "Waiting for pool to start... Current state: $($poolState.Value)"
                                    }
                                    
                                    if ($poolState.Value -eq "Started") {
                                        Write-Host "Application pool started successfully!"
                                        return $true
                                    } else {
                                        Write-Host "Warning: Application pool state is $($poolState.Value)" -ForegroundColor Yellow
                                        return $false
                                    }
                                } else {
                                    Write-Host "Warning: Application pool '$appPoolName' not found" -ForegroundColor Yellow
                                    return $true
                                }
                            } catch {
                                Write-Host "Warning: Could not start application pool: $_" -ForegroundColor Yellow
                                return $false
                            }
                        }
                        
                        # Function to stop website
                        function Stop-IISWebsite {
                            param($websiteName)
                            
                            try {
                                Import-Module WebAdministration -ErrorAction Stop
                                
                                if (Test-Path "IIS:\\Sites\\$websiteName") {
                                    Write-Host "Stopping website: $websiteName"
                                    Stop-Website -Name $websiteName -ErrorAction Stop
                                    Write-Host "Website stopped successfully!"
                                    return $true
                                } else {
                                    Write-Host "Warning: Website '$websiteName' not found" -ForegroundColor Yellow
                                    return $true
                                }
                            } catch {
                                Write-Host "Warning: Could not stop website: $_" -ForegroundColor Yellow
                                
                                # Try using appcmd as fallback
                                try {
                                    Write-Host "Attempting to stop website using appcmd..."
                                    & appcmd stop site /site.name:"$websiteName" 2>&1
                                    Start-Sleep -Seconds 5
                                    return $true
                                } catch {
                                    return $false
                                }
                            }
                        }
                        
                        # Function to start website
                        function Start-IISWebsite {
                            param($websiteName)
                            
                            try {
                                if (Test-Path "IIS:\\Sites\\$websiteName") {
                                    Write-Host "Starting website: $websiteName"
                                    Start-Website -Name $websiteName -ErrorAction Stop
                                    Write-Host "Website started successfully!"
                                    return $true
                                } else {
                                    Write-Host "Warning: Website '$websiteName' not found" -ForegroundColor Yellow
                                    return $true
                                }
                            } catch {
                                Write-Host "Warning: Could not start website: $_" -ForegroundColor Yellow
                                return $false
                            }
                        }
                        
                        # Function to release file locks
                        function Release-FileLocks {
                            param($path)
                            
                            Write-Host "Attempting to release file locks in: $path"
                            
                            # Force garbage collection
                            [System.GC]::Collect()
                            [System.GC]::WaitForPendingFinalizers()
                            
                            # Try to find and kill processes holding locks
                            try {
                                $files = Get-ChildItem -Path $path -Recurse -ErrorAction SilentlyContinue
                                foreach ($file in $files) {
                                    try {
                                        # Try to open file to see if it's locked
                                        $fileStream = [System.IO.File]::Open($file.FullName, 'Open', 'Read', 'None')
                                        $fileStream.Close()
                                        $fileStream.Dispose()
                                    } catch {
                                        Write-Host "File is locked: $($file.Name)"
                                    }
                                }
                            } catch {
                                # Ignore errors
                            }
                            
                            Start-Sleep -Seconds 5
                        }
                        
                        # --- MAIN DEPLOYMENT PROCESS ---
                        
                        # Check if source publish folder exists
                        if (-not (Test-Path $publishFolder)) {
                            throw "Source publish folder not found!"
                        }
                        
                        # Step 1: Stop IIS services
                        Write-Host "`n========================================"
                        Write-Host "Step 1: Stopping IIS Services"
                        Write-Host "========================================"
                        
                        # Stop website first
                        $websiteStopped = Stop-IISWebsite -websiteName $websiteName
                        
                        # Then stop app pool with force
                        $poolStopped = Stop-IISAppPoolForce -appPoolName $appPoolName
                        
                        # Wait for locks to release
                        Write-Host "Waiting $lockReleaseWait seconds for file locks to be released..."
                        Start-Sleep -Seconds $lockReleaseWait
                        
                        # Step 2: Deploy files
                        Write-Host "`n========================================"
                        Write-Host "Step 2: Deploying Files"
                        Write-Host "========================================"
                        
                        # Check destination
                        try {
                            if (-not (Test-Path $destination)) {
                                Write-Host "Destination folder does not exist. Creating it..."
                                New-Item -ItemType Directory -Path $destination -Force | Out-Null
                                Write-Host "Destination folder created successfully!"
                            } else {
                                Write-Host "Destination folder exists and is accessible."
                            }
                        } catch {
                            Write-Host "Cannot access destination folder: $_" -ForegroundColor Red
                            throw "Destination folder is not accessible. Please check permissions."
                        }
                        
                        # Clean destination folder - ignore errors on locked files
                        Write-Host "Cleaning destination folder..."
                        $cleanSuccess = $false
                        $cleanRetry = 0
                        
                        while (-not $cleanSuccess -and $cleanRetry -lt $maxRetries) {
                            try {
                                if (Test-Path $destination) {
                                    # Remove read-only attributes
                                    Get-ChildItem -Path $destination -Recurse -Force -ErrorAction SilentlyContinue | ForEach-Object {
                                        try {
                                            if ($_.IsReadOnly) {
                                                $_.IsReadOnly = $false
                                            }
                                        } catch {
                                            # Ignore errors for individual items
                                        }
                                    }
                                    
                                    # Try to delete files - if locked, ignore and continue
                                    Get-ChildItem -Path $destination -Recurse -Force -ErrorAction SilentlyContinue | ForEach-Object {
                                        try {
                                            Remove-Item -Path $_.FullName -Force -ErrorAction Stop
                                        } catch {
                                            Write-Host "Warning: Could not delete $($_.Name): $_" -ForegroundColor Yellow
                                            # Continue with next file
                                        }
                                    }
                                    
                                    # Try to delete empty folders
                                    Get-ChildItem -Path $destination -Recurse -Force -ErrorAction SilentlyContinue | Sort-Object -Property FullName -Descending | ForEach-Object {
                                        try {
                                            if ($_.PSIsContainer) {
                                                Remove-Item -Path $_.FullName -Force -ErrorAction SilentlyContinue
                                            }
                                        } catch {
                                            # Ignore
                                        }
                                    }
                                    
                                    $cleanSuccess = $true
                                    Write-Host "Destination folder cleaned successfully!"
                                }
                            } catch {
                                $cleanRetry++
                                Write-Host "Clean attempt $cleanRetry had some issues: $_" -ForegroundColor Yellow
                                if ($cleanRetry -lt $maxRetries) {
                                    Write-Host "Waiting 5 seconds before retry..."
                                    Start-Sleep -Seconds 5
                                    [System.GC]::Collect()
                                    [System.GC]::WaitForPendingFinalizers()
                                }
                            }
                        }
                        
                        if (-not $cleanSuccess) {
                            Write-Host "Warning: Could not fully clean destination folder. Some files may be locked." -ForegroundColor Yellow
                        }
                        
                        # Copy files - ignore locked files
                        Write-Host "Copying files from publish folder..."
                        $retryCount = 0
                        $copied = $false
                        
                        while (-not $copied -and $retryCount -lt $maxRetries) {
                            try {
                                # Copy all files, overwriting existing
                                Copy-Item -Path "$publishFolder\\*" -Destination $destination -Recurse -Force -ErrorAction Continue
                                $copied = $true
                                Write-Host "Files copied successfully (some errors may have been ignored)!"
                            } catch {
                                $retryCount++
                                Write-Host "Copy attempt $retryCount had errors: $_" -ForegroundColor Yellow
                                if ($retryCount -lt $maxRetries) {
                                    Write-Host "Retrying in 10 seconds..."
                                    Start-Sleep -Seconds 10
                                    [System.GC]::Collect()
                                    [System.GC]::WaitForPendingFinalizers()
                                }
                            }
                        }
                        
                        if (-not $copied) {
                            Write-Host "Warning: Could not copy all files after $maxRetries attempts" -ForegroundColor Yellow
                            Write-Host "Some files may still be locked." -ForegroundColor Yellow
                            Write-Host "Deployment may be incomplete." -ForegroundColor Yellow
                        }
                        
                        # Step 3: Start IIS services
                        Write-Host "`n========================================"
                        Write-Host "Step 3: Starting IIS Services"
                        Write-Host "========================================"
                        
                        $poolStarted = Start-IISAppPool -appPoolName $appPoolName
                        $websiteStarted = Start-IISWebsite -websiteName $websiteName
                        
                        # Verify deployment
                        Write-Host "`n========================================"
                        Write-Host "Step 4: Verification"
                        Write-Host "========================================"
                        
                        try {
                            $sourceCount = (Get-ChildItem -Path $publishFolder -Recurse -Force -ErrorAction SilentlyContinue | Measure-Object).Count
                            $destCount = (Get-ChildItem -Path $destination -Recurse -Force -ErrorAction SilentlyContinue | Measure-Object).Count
                            
                            Write-Host "Deployment Summary:"
                            Write-Host "Source files: $sourceCount"
                            Write-Host "Destination files: $destCount"
                            Write-Host "Destination path: $destination"
                            Write-Host "Application Pool: $appPoolName"
                            Write-Host "Website: $websiteName"
                            Write-Host "========================================"
                            
                            if ($sourceCount -eq $destCount) {
                                Write-Host "Deployment completed successfully!" -ForegroundColor Green
                            } else {
                                Write-Host "Warning: File count mismatch. Deployment may be incomplete." -ForegroundColor Yellow
                                Write-Host "This might be normal if some files were locked." -ForegroundColor Yellow
                            }
                        } catch {
                            Write-Host "Warning: Could not verify deployment: $_" -ForegroundColor Yellow
                        }
                        
                        # Consider deployment successful even with warnings
                        Write-Host "`nDeployment process completed!" -ForegroundColor Green
                    '''
                }
            }
        }
        
        stage('CopyConfig') {
            steps {
                script {
                    powershell '''
                        Write-Host "========================================"
                        Write-Host "Copying Configuration Files"
                        Write-Host "========================================"
                        
                        $configSource = "D:\\IIS_Root\\RI_TRACKING_DEV\\RI_TRACKING_CONFIG\\ri_tracking_backend_config\\appsettings.json"
                        $configDestination = "D:\\IIS_Root\\RI_TRACKING_DEV\\RI_TRACKING_BACKEND"
                        $destFile = Join-Path $configDestination "appsettings.json"
                        
                        # Check if source file exists
                        if (Test-Path $configSource) {
                            Write-Host "Source file found: $configSource"
                            
                            # Check if destination path is accessible
                            try {
                                if (-not (Test-Path $configDestination)) {
                                    Write-Host "Destination directory does not exist. Creating it..."
                                    New-Item -ItemType Directory -Path $configDestination -Force | Out-Null
                                    Write-Host "Destination directory created successfully!"
                                }
                            } catch {
                                Write-Host "Cannot access destination path: $_" -ForegroundColor Red
                                throw "Destination path is not accessible. Please check permissions."
                            }
                            
                            # Delete existing appsettings if it exists
                            if (Test-Path $destFile) {
                                Write-Host "Deleting existing appsettings at destination..."
                                try {
                                    $existingFile = Get-Item $destFile -ErrorAction Stop
                                    if ($existingFile.IsReadOnly) {
                                        $existingFile.IsReadOnly = $false
                                        Write-Host "Removed read-only attribute from existing file"
                                    }
                                    Remove-Item -Path $destFile -Force -ErrorAction Stop
                                    Write-Host "Existing appsettings deleted successfully!"
                                } catch {
                                    Write-Host "Warning: Could not delete existing file: $_" -ForegroundColor Yellow
                                    # Try to overwrite instead
                                }
                            } else {
                                Write-Host "No existing appsettings found at destination"
                            }
                            
                            # Copy config file - ignore errors and continue
                            Write-Host "Copying appsettings to $configDestination"
                            try {
                                Copy-Item -Path $configSource -Destination $configDestination -Force -ErrorAction Stop
                                Write-Host "Config file copied successfully!"
                            } catch {
                                Write-Host "Warning: Could not copy config file: $_" -ForegroundColor Yellow
                                Write-Host "Config file may need to be copied manually." -ForegroundColor Yellow
                            }
                            
                            # Verify the file was copied
                            if (Test-Path $destFile) {
                                Write-Host "Verification: appsettings successfully copied to destination"
                                Write-Host "File size: $((Get-Item $destFile).Length) bytes"
                                Write-Host "File path: $destFile"
                                Write-Host "Config Copy Completed Successfully!"
                            } else {
                                Write-Host "Warning: appsettings not found in destination after copy" -ForegroundColor Yellow
                                Write-Host "Config file may need to be copied manually." -ForegroundColor Yellow
                            }
                        } else {
                            Write-Host "Warning: Source config file not found at: $configSource" -ForegroundColor Yellow
                            Write-Host "Skipping config copy." -ForegroundColor Yellow
                        }
                        
                        Write-Host "========================================"
                    '''
                }
            }
        }
    }

    post {
        success {
            echo 'Pipeline executed successfully!'
        }
        failure {
            echo 'Pipeline failed!'
            script {
                // Try to restart IIS services even on failure
                powershell '''
                    Write-Host "Attempting to restart IIS services after failure..."
                    $appPoolName = $env:APP_POOL_NAME
                    $websiteName = $env:WEBSITE_NAME
                    
                    try {
                        Import-Module WebAdministration -ErrorAction SilentlyContinue
                        if (Test-Path "IIS:\\Sites\\$websiteName") {
                            Start-Website -Name $websiteName -ErrorAction SilentlyContinue
                        }
                        if (Test-Path "IIS:\\AppPools\\$appPoolName") {
                            Start-WebAppPool -Name $appPoolName -ErrorAction SilentlyContinue
                        }
                        Write-Host "IIS services restart attempted."
                    } catch {
                        Write-Host "Could not restart IIS services: $_"
                    }
                '''
            }
        }
        always {
            script {
                // Clean up publish folder
                bat "if exist .\\publish rmdir /s /q .\\publish"
            }
        }
    }
}
