pipeline {
    agent any
    environment {
        SOURCE_FILE = "${WORKSPACE}\\appsettings.json"
        TARGET_DIR  = "D:\\IIS_Root\\env"
        TARGET_FILE = "D:\\IIS_Root\\env\\appsettings.json"
    }
    stages {
        stage('Checkout') {
            steps { checkout scm }   // ✅ now valid
        }
        stage('Validate JSON') {
            steps {
                powershell '''
                    $ErrorActionPreference = "Stop"
                    if (-not (Test-Path $env:SOURCE_FILE)) {
                        throw "appsettings.json not found at $env:SOURCE_FILE"
                    }
                    Get-Content $env:SOURCE_FILE -Raw | ConvertFrom-Json | Out-Null
                    Write-Host "✅ JSON is valid"
                '''
            }
        }
        stage('Deploy to D:\\IIS_Root\\env') {
            steps {
                powershell '''
                    $ErrorActionPreference = "Stop"
                    if (-not (Test-Path $env:TARGET_DIR)) {
                        New-Item -ItemType Directory -Path $env:TARGET_DIR -Force | Out-Null
                    }
                    if (Test-Path $env:TARGET_FILE) {
                        $ts = Get-Date -Format "yyyyMMdd_HHmmss"
                        Copy-Item $env:TARGET_FILE "$env:TARGET_FILE.$ts.bak" -Force
                    }
                    Copy-Item $env:SOURCE_FILE $env:TARGET_FILE -Force
                    Write-Host "✅ Published to $env:TARGET_FILE"
                '''
            }
        }
    }
}
